#fact_sales
SELECT
    id AS sales_id,
    sales_date,
    store_nbr,
    family,
    sales,
    onpromotion,
    city,
    state,
    type AS store_type,
    cluster,
    oil_price,
    holiday_type,
    holiday_locale,
    holiday_locale_name,
    holiday_description,
    holiday_transferred
FROM {{ ref('int_sales_complete') }}

#dim_date
SELECT DISTINCT
    sales_date AS date,
    YEAR(sales_date) AS year,
    MONTH(sales_date) AS month,
    DAY(sales_date) AS day,
    DAYOFWEEK(sales_date) AS day_of_week
FROM {{ ref('int_sales_complete') }}
WHERE sales_date IS NOT NULL

#dim_product
SELECT DISTINCT
    family AS product_family
FROM {{ ref('int_train_sales') }}
WHERE family IS NOT NULL

#dim_store
SELECT DISTINCT
    store_nbr,
    city,
    state,
    type AS store_type,
    cluster
FROM {{ ref('stg_stores') }}
WHERE store_nbr IS NOT NULL

#agg_daily_sales
SELECT
    sales_date AS date,
    COUNT(*) AS sales_records,
    COUNT(DISTINCT store_nbr) AS total_stores,
    COUNT(DISTINCT family) AS total_product_families,
    SUM(sales) AS total_sales,
    AVG(sales) AS average_sales,
    SUM(onpromotion) AS total_promotions
FROM {{ ref('fact_sales') }}
GROUP BY sales_date
ORDER BY sales_date

#agg_daily_sales
SELECT
    sales_date AS date,
    holiday_type,
    holiday_locale,
    holiday_locale_name,
    COUNT(DISTINCT store_nbr) AS total_stores,
    COUNT(DISTINCT family) AS total_product_families,
    SUM(sales) AS total_sales,
    AVG(sales) AS average_sales,
    SUM(onpromotion) AS total_promotions
FROM {{ ref('fact_sales') }}
WHERE holiday_type <> 'No Holiday'
GROUP BY
    sales_date,
    holiday_type,
    holiday_locale,
    holiday_locale_name
ORDER BY sales_date;

#agg_monthly_sales

SELECT
    YEAR(sales_date) AS year,
    MONTH(sales_date) AS month,
    COUNT(*) AS sales_records,
    COUNT(DISTINCT store_nbr) AS total_stores,
    COUNT(DISTINCT family) AS total_product_families,
    SUM(sales) AS total_sales,
    AVG(sales) AS average_sales,
    SUM(onpromotion) AS total_promotions
FROM {{ ref('fact_sales') }}
GROUP BY
    YEAR(sales_date),
    MONTH(sales_date)
ORDER BY
    year,
    month

#agg_product_performance

SELECT
    family AS product_family,
    COUNT(*) AS sales_records,
    COUNT(DISTINCT sales_date) AS active_days,
    COUNT(DISTINCT store_nbr) AS total_stores,
    SUM(sales) AS total_sales,
    AVG(sales) AS average_sales,
    SUM(onpromotion) AS total_promotions
FROM {{ ref('fact_sales') }}
GROUP BY family
ORDER BY total_sales DESC


#agg_promotion_sales

SELECT
    family AS product_family,
    COUNT(*) AS sales_records,
    COUNT(DISTINCT sales_date) AS active_days,
    COUNT(DISTINCT store_nbr) AS total_stores,
    SUM(sales) AS total_sales,
    AVG(sales) AS average_sales,
    SUM(onpromotion) AS total_promotions
FROM {{ ref('fact_sales') }}
GROUP BY family
ORDER BY total_sales DESC

#agg_store_performance

SELECT
    store_nbr,
    city,
    state,
    store_type,
    cluster,
    COUNT(*) AS sales_records,
    COUNT(DISTINCT sales_date) AS active_days,
    COUNT(DISTINCT family) AS product_families,
    SUM(sales) AS total_sales,
    AVG(sales) AS average_sales,
    SUM(onpromotion) AS total_promotions
FROM {{ ref('fact_sales') }}
GROUP BY
    store_nbr,

    state,
